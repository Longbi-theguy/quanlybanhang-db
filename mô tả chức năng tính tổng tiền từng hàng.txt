-- Câu 5: Tính tổng tiền từng đơn hàng

-- Nghiệp vụ : Tính giá trị thanh toán của từng đơn dựa trên số lượng và đơn giá.

-- Quan hệ   : Gom nhóm tính toán trên bảng [OrderDetails].

-- -----------------------------------------------------------------------------

SELECT 

    OrderID,

    SUM(Quantity * UnitPrice) AS TotalAmount

FROM OrderDetails

GROUP BY OrderID;

GO
