-- -----------------------------------------------------------------------------

-- Câu 7: Tổng doanh thu theo từng sản phẩm (Chỉ tính đơn Completed)

-- Nghiệp vụ : Đánh giá hiệu quả kinh doanh của từng mặt hàng.

-- Quan hệ   : Nối 3 bảng [Products] -> [OrderDetails] -> [Orders].

-- -----------------------------------------------------------------------------

SELECT 

    p.ProductID,

    p.ProductName,

    SUM(od.Quantity * od.UnitPrice) AS TotalRevenue

FROM Products p

JOIN OrderDetails od ON p.ProductID = od.ProductID

JOIN Orders o        ON od.OrderID = o.OrderID

WHERE o.Status = 'Completed'

GROUP BY 

    p.ProductID,

    p.ProductName;

GO
