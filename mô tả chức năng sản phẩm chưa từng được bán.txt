-- Câu 11: Sản phẩm chưa từng được bán

-- Nghiệp vụ : Phát hiện hàng tồn kho ế ẩm để lên kế hoạch giải phóng tồn kho.

-- Quan hệ   : Subquery đối chiếu giữa [Products] và [OrderDetails].

-- -----------------------------------------------------------------------------

SELECT 

    ProductID,

    ProductName,

    Price,

    Stock

FROM Products

WHERE ProductID NOT IN (

    SELECT DISTINCT ProductID 

    FROM OrderDetails

);

GO
