-- -----------------------------------------------------------------------------

-- Câu 2: Sản phẩm cao cấp (Giá > 1.000.000 VNĐ)

-- Nghiệp vụ : Lọc sản phẩm phân khúc cao để chạy chương trình ưu đãi.

-- Quan hệ   : Thao tác trên bảng đơn [Products].

-- -----------------------------------------------------------------------------

SELECT 

    ProductID,

    ProductName,

    Price,

    Stock

FROM Products

WHERE Price > 1000000;

GO
