-- -----------------------------------------------------------------------------

-- Câu 12: Khách hàng thân thiết (Từ 2 đơn hàng trở lên)

-- Nghiệp vụ : Đo lường tỷ lệ quay lại mua hàng (Retention Rate).

-- Quan hệ   : Nối [Customers] với [Orders] (1:N) kết hợp HAVING.

-- -----------------------------------------------------------------------------

SELECT 

    c.CustomerID,

    c.CustomerName,

    COUNT(o.OrderID) AS OrderCount

FROM Customers c

JOIN Orders o ON c.CustomerID = o.CustomerID

GROUP BY 

    c.CustomerID,

    c.CustomerName

HAVING COUNT(o.OrderID) >= 2;

GO
